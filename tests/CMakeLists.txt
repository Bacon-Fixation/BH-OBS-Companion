cmake_minimum_required(VERSION 3.16)
project(bh-native-language-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(native-detector-test native-detector-test.cpp ../src/native-language-detector.cpp)
target_include_directories(native-detector-test PRIVATE ../src)
target_link_libraries(native-detector-test PRIVATE Qt6::Core)
enable_testing()
add_test(NAME native-language-parity COMMAND native-detector-test ${CMAKE_CURRENT_SOURCE_DIR}/../data/language-guesser ${CMAKE_CURRENT_SOURCE_DIR}/native-fixtures.json)

set(CMAKE_AUTOMOC ON)
add_executable(native-worker-test native-worker-test.cpp ../src/language-feedback.cpp ../src/native-language-worker.cpp ../src/native-language-worker.hpp ../src/native-language-detector.cpp)
target_include_directories(native-worker-test PRIVATE ../src)
target_compile_definitions(native-worker-test PRIVATE QT_NO_KEYWORDS=1)
target_link_libraries(native-worker-test PRIVATE Qt6::Core)
add_test(NAME native-worker-lifecycle COMMAND native-worker-test ${CMAKE_CURRENT_SOURCE_DIR}/../data/language-guesser)

add_executable(native-model-test native-model-test.cpp ../src/language-feedback.cpp ../src/native-language-worker.cpp ../src/native-language-worker.hpp ../src/native-language-detector.cpp)
target_include_directories(native-model-test PRIVATE ../src)
target_compile_definitions(native-model-test PRIVATE QT_NO_KEYWORDS=1)
target_link_libraries(native-model-test PRIVATE Qt6::Core)
add_test(NAME native-model-management COMMAND native-model-test ${CMAKE_CURRENT_SOURCE_DIR}/../data/language-guesser)

add_executable(native-confidence-test native-confidence-test.cpp ../src/native-language-detector.cpp)
target_include_directories(native-confidence-test PRIVATE ../src)
target_link_libraries(native-confidence-test PRIVATE Qt6::Core)
add_test(NAME native-confidence-evidence COMMAND native-confidence-test ${CMAKE_CURRENT_SOURCE_DIR}/../data/language-guesser)

add_executable(native-detect-jsonl native-detect-jsonl.cpp ../src/native-language-detector.cpp)
target_include_directories(native-detect-jsonl PRIVATE ../src)
target_link_libraries(native-detect-jsonl PRIVATE Qt6::Core)
